/**
 * MM Testimonials — admin panel + shortcode [mm_testimonials]
 * Content:  WP Admin → Settings → Testimonials
 * Styling:  the "--ts-" CSS snippet
 */

define( 'MM_TS_CARDS', 4 );

/* ── the heading shown if the admin field is left empty ── */
define( 'MM_TS_FALLBACK_HEADING', 'Testimonials' );

function mm_ts_defaults() {
	return array(
		'eyebrow' => '',
		'line1'   => 'Testimonials',
		'line2'   => '',
		'hint'    => 'Tap to read',
		'cards'   => array(
			array( 'img' => '', 'rating' => '4.9', 'quote' => 'The dining chairs arrived earlier than promised and the finish is even better in person.', 'name' => 'Ananya Kapoor', 'role' => 'Verified buyer' ),
			array( 'img' => '', 'rating' => '5.0', 'quote' => 'I ordered the bedroom chair on a whim and it is now my favourite corner of the house.',      'name' => 'Rohan Mehta',   'role' => 'Verified buyer' ),
			array( 'img' => '', 'rating' => '4.7', 'quote' => 'Packaging was spotless and assembly took ten minutes. Support replied the same evening.',      'name' => 'Priya Nair',    'role' => 'Verified buyer' ),
			array( 'img' => '', 'rating' => '4.9', 'quote' => 'Bought a full dining set. Three months in and it still looks brand new.',                      'name' => 'Karan Singh',   'role' => 'Verified buyer' ),
		),
	);
}

function mm_ts_get() {
	$saved = get_option( 'mm_ts_opts', array() );
	$d     = mm_ts_defaults();
	$out   = wp_parse_args( is_array( $saved ) ? $saved : array(), $d );

	for ( $i = 0; $i < MM_TS_CARDS; $i++ ) {
		$out['cards'][ $i ] = wp_parse_args(
			isset( $out['cards'][ $i ] ) && is_array( $out['cards'][ $i ] ) ? $out['cards'][ $i ] : array(),
			$d['cards'][ $i ]
		);
	}
	return $out;
}

add_action( 'admin_menu', function () {
	add_options_page( 'Testimonials', 'Testimonials', 'manage_options', 'mm-testimonials', 'mm_ts_page' );
} );

add_action( 'admin_init', function () {
	register_setting( 'mm_ts_group', 'mm_ts_opts', array( 'sanitize_callback' => 'mm_ts_sanitize' ) );
} );

function mm_ts_sanitize( $in ) {
	$out = array(
		'eyebrow' => sanitize_text_field( $in['eyebrow'] ?? '' ),
		'line1'   => sanitize_text_field( $in['line1']   ?? '' ),
		'line2'   => sanitize_text_field( $in['line2']   ?? '' ),
		'hint'    => sanitize_text_field( $in['hint']    ?? '' ),
		'cards'   => array(),
	);

	for ( $i = 0; $i < MM_TS_CARDS; $i++ ) {
		$c = $in['cards'][ $i ] ?? array();
		$r = isset( $c['rating'] ) ? (float) $c['rating'] : 5;
		$r = max( 0, min( 5, $r ) );

		$out['cards'][ $i ] = array(
			'img'    => esc_url_raw( $c['img'] ?? '' ),
			'rating' => number_format( $r, 1 ),
			'quote'  => sanitize_textarea_field( $c['quote'] ?? '' ),
			'name'   => sanitize_text_field( $c['name'] ?? '' ),
			'role'   => sanitize_text_field( $c['role'] ?? '' ),
		);
	}
	return $out;
}

add_action( 'admin_enqueue_scripts', function ( $hook ) {
	if ( $hook === 'settings_page_mm-testimonials' ) { wp_enqueue_media(); }
} );

function mm_ts_stars( $rating ) {
	$full = (int) max( 0, min( 5, floor( (float) $rating + 0.25 ) ) );
	return str_repeat( '★', $full ) . str_repeat( '☆', 5 - $full );
}

/* ── the shortcode ── */
function mm_testimonials_shortcode() {
	$o   = mm_ts_get();
	$sid = 'mmts' . wp_rand( 1000, 9999 );

	/* heading can never come out empty */
	$line1 = trim( $o['line1'] );
	if ( $line1 === '' ) { $line1 = MM_TS_FALLBACK_HEADING; }
	$line2 = trim( $o['line2'] );

	ob_start(); ?>
	<section class="ts" id="<?php echo esc_attr( $sid ); ?>">

		<div class="ts-head">
			<?php if ( trim( $o['eyebrow'] ) !== '' ) : ?>
				<p class="ts-eyebrow"><?php echo esc_html( $o['eyebrow'] ); ?></p>
			<?php endif; ?>
			<h2 class="ts-title">
				<span><?php echo esc_html( $line1 ); ?></span>
				<?php if ( $line2 !== '' ) : ?>
					<span><?php echo esc_html( $line2 ); ?></span>
				<?php endif; ?>
			</h2>
		</div>

		<div class="ts-viewport">
			<div class="ts-row">
				<?php for ( $i = 0; $i < MM_TS_CARDS; $i++ ) :
					$c  = $o['cards'][ $i ];
					$id = $sid . '-' . $i;
					$bg = $c['img'] ? 'url(' . esc_url( $c['img'] ) . ')' : 'none';
					?>
					<div class="ts-card" style="--img:<?php echo $bg; ?>;">
						<input type="checkbox" class="ts-toggle" id="<?php echo esc_attr( $id ); ?>">
						<label class="ts-flip" for="<?php echo esc_attr( $id ); ?>">
							<span class="ts-inner">

								<span class="ts-front">
									<?php if ( trim( $o['hint'] ) !== '' ) : ?>
										<span class="ts-hint"><?php echo esc_html( $o['hint'] ); ?></span>
									<?php endif; ?>
								</span>

								<span class="ts-back">
									<span class="ts-top">
										<span class="ts-stars"><?php echo esc_html( mm_ts_stars( $c['rating'] ) ); ?></span>
										<span class="ts-rate"><?php echo esc_html( $c['rating'] ); ?></span>
									</span>
									<span class="ts-quote"><?php echo esc_html( $c['quote'] ); ?></span>
									<span class="ts-person">
										<span class="ts-avatar"></span>
										<span class="ts-who">
											<span class="ts-name"><?php echo esc_html( $c['name'] ); ?></span>
											<span class="ts-role"><?php echo esc_html( $c['role'] ); ?></span>
										</span>
									</span>
								</span>

							</span>
						</label>
					</div>
				<?php endfor; ?>
			</div>
		</div>

		<div class="ts-nav">
			<button type="button" class="ts-arrow ts-prev" aria-label="Previous">&#8249;</button>
			<span class="ts-dots"></span>
			<button type="button" class="ts-arrow ts-next" aria-label="Next">&#8250;</button>
		</div>

	</section>

	<script>
	(function(){
		function init(){
			var root = document.getElementById('<?php echo esc_js( $sid ); ?>');
			if(!root || root.getAttribute('data-mm-init')) return;
			root.setAttribute('data-mm-init','1');

			var row   = root.querySelector('.ts-row');
			var cards = Array.prototype.slice.call(root.querySelectorAll('.ts-card'));
			var prev  = root.querySelector('.ts-prev');
			var next  = root.querySelector('.ts-next');
			var dotsB = root.querySelector('.ts-dots');
			var i = 0, moved = false;

			cards.forEach(function(c, n){
				var d = document.createElement('button');
				d.type = 'button';
				d.className = 'ts-dot';
				d.setAttribute('aria-label','Go to card ' + (n+1));
				d.addEventListener('click', function(){ go(n); });
				dotsB.appendChild(d);
			});
			var dots = Array.prototype.slice.call(dotsB.children);

			function isMobile(){ return window.matchMedia('(max-width: 900px)').matches; }

			function step(){
				if(!cards.length) return 0;
				var cs  = window.getComputedStyle(row);
				var gap = parseFloat(cs.columnGap || cs.gap) || 0;
				return cards[0].getBoundingClientRect().width + gap;
			}

			function go(n){
				i = Math.max(0, Math.min(cards.length - 1, n));
				if(isMobile()){
					row.style.transform = 'translateX(' + (-step() * i) + 'px)';
				} else {
					row.style.transform = '';
				}
				dots.forEach(function(d, x){ d.className = 'ts-dot' + (x === i ? ' is-on' : ''); });
				if(prev) prev.disabled = (i === 0);
				if(next) next.disabled = (i === cards.length - 1);
			}

			if(prev) prev.addEventListener('click', function(){ go(i - 1); });
			if(next) next.addEventListener('click', function(){ go(i + 1); });

			var x0 = null, y0 = null;
			row.addEventListener('touchstart', function(e){
				x0 = e.touches[0].clientX; y0 = e.touches[0].clientY; moved = false;
			}, {passive:true});

			row.addEventListener('touchmove', function(e){
				if(x0 === null) return;
				var dx = e.touches[0].clientX - x0;
				var dy = e.touches[0].clientY - y0;
				if(Math.abs(dx) > 12 && Math.abs(dx) > Math.abs(dy)) moved = true;
			}, {passive:true});

			row.addEventListener('touchend', function(e){
				if(x0 === null) return;
				var dx = e.changedTouches[0].clientX - x0;
				if(moved && Math.abs(dx) > 40){ go(i + (dx < 0 ? 1 : -1)); }
				x0 = null;
			}, {passive:true});

			row.addEventListener('click', function(e){
				if(moved){ e.preventDefault(); e.stopPropagation(); moved = false; }
			}, true);

			var t;
			window.addEventListener('resize', function(){
				clearTimeout(t);
				t = setTimeout(function(){ go(i); }, 120);
			});

			go(0);
		}
		if(document.readyState !== 'loading'){ init(); }
		else { document.addEventListener('DOMContentLoaded', init); }
	})();
	</script>
	<?php
	return ob_get_clean();
}
add_shortcode( 'mm_testimonials', 'mm_testimonials_shortcode' );
